{{- /* Imágenes de markdown (![alt](/images/x.png)): carga diferida y, si el fichero está en static/, ancho y alto para que la página no salte al cargar. Para figuras procesadas a webp usa el shortcode img. Si el fichero es un .mp4 (un GIF convertido con scripts/gif2mp4.swift) se pinta como vídeo en bucle; su póster es el .png del mismo nombre y partials/loop_video.html lo reproduce al entrar en pantalla. */ -}} {{- $src := .Destination -}} {{- $isVideo := strings.HasSuffix (lower $src) ".mp4" -}} {{- $sized := cond $isVideo (replaceRE `(?i)\.mp4$` ".png" $src) $src -}} {{- $w := 0 }}{{ $h := 0 -}} {{- if hasPrefix $sized "/" -}} {{- $file := printf "static%s" $sized -}} {{- if fileExists $file }}{{ with imageConfig $file }}{{ $w = .Width }}{{ $h = .Height }}{{ end }}{{ end -}} {{- end -}} {{- if $isVideo -}} {{- .Page.Store.Set "hasLoopVideo" true -}} {{- else -}} {{- end -}} {{- /* sin salto de línea final: Hugo lo metería dentro del párrafo */ -}}
